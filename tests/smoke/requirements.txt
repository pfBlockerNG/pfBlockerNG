# ADR-04 smoke-harness dev dependencies — DEV ONLY, never shipped.
#
# The release archive contains only src/; this file (like
# benchmarks/requirements.txt, ADR-01) is part of the dev tooling and is not
# packaged. Install on the smoke runner with:
#   python -m pip install -r tests/smoke/requirements.txt
#
# Kept deliberately small: the VM lifecycle (boot/overlay/teardown) and the SSH
# round-trip reuse the existing POSIX-sh helpers (tests/smoke/boot_vm.sh,
# wait_ready.sh) over subprocess + the system `ssh`/`qemu` binaries, so no
# Python QEMU/SSH library is needed. The mock feed server is stdlib http.server
# only. The one third-party dep is dnspython, for asserting DNS rcode+record
# against the guest's real Unbound from Python (richer + more precise than
# shelling out to `dig`).
#
# These modules are imported ONLY by tests/smoke/* , which the default
# `python -m pytest` ignores (pyproject.toml addopts: --ignore=tests/smoke), so
# a default run on a host WITHOUT these installed still collects cleanly.

dnspython==2.8.0
# Per-test wall-clock ceiling so a wedged VM op (SSH/pkg/dig) fails that test
# fast instead of hanging the job to its timeout-minutes cap.
pytest-timeout==2.4.0
# ADR-14 Web-UI tier: the authenticated webConfigurator session client
# (tests/smoke/ui/webui.py) drives the CSRF form login + page GETs over HTTP.
# Imported lazily (inside WebUI), so default collection -- which --ignore's
# tests/smoke -- never needs it.
# >=2.33.0 fixes CVE-2026-25645 (predictable tmp filenames in extract_zipped_paths).
requests==2.34.2
# ADR-14 Tier-B (browser): headless Chromium drives the JS-only UX
# (tests/smoke/ui/test_browser.py) + captures per-page screenshot artifacts.
# Imported lazily via pytest.importorskip (conftest + the test module), so a host
# without it still collects tests/smoke/ui cleanly (the browser tests SKIP).
# NOTE: the pip install gives the Python bindings only -- the Chromium BROWSER
# BINARY is a separate download (`python -m playwright install chromium`), wired
# as a CI setup step in Phase 5; do NOT assume it is present from this line alone.
# >=1.60.0 loosens its greenlet dep to a range (greenlet>=3.1.1,<4) so it
# resolves a greenlet with cp313/cp314 wheels (3.1.1 had none and broke the
# 26.04 runner's newer-CPython build).
playwright==1.60.0
